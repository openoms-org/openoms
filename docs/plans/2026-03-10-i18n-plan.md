# Internationalization (i18n) Implementation Plan

> **For Claude:** Implement tasks sequentially. After each task: verify build, commit. Branch: `feat/i18n` (already created in public repo).

**Goal:** Make OpenOMS multi-language with auto-detection from browser, user-saveable preference, English + Polish on launch.

**Architecture:** `next-intl` (no URL locale) on frontend with cookie-based locale. Backend becomes language-agnostic — returns keys instead of Polish text. User preference stored in DB (`users.language`) + localStorage/cookie.

**Tech Stack:** Go 1.25, pgx/v5, Next.js 16, next-intl, Zustand, Zod v4

**Conventions:**
- No Co-Authored-By or attribution in commits
- Code in English, user-facing translations in JSON files
- Always branch + PR, never push to main

### 6b: Create i18n request config

Create `src/i18n/request.ts`:
```typescript
import { getRequestConfig } from "next-intl/server";
import { cookies, headers } from "next/headers";

const SUPPORTED_LOCALES = ["en", "pl"];
const DEFAULT_LOCALE = "en";

function detectLocale(acceptLanguage: string | null): string {
  if (!acceptLanguage) return DEFAULT_LOCALE;
  const preferred = acceptLanguage.split(",").map(l => l.split(";")[0].trim().split("-")[0]);
  return preferred.find(l => SUPPORTED_LOCALES.includes(l)) ?? DEFAULT_LOCALE;
}

export default getRequestConfig(async () => {
  const cookieStore = await cookies();
  const headerStore = await headers();

  // Priority: cookie > Accept-Language > default
  const cookieLocale = cookieStore.get("NEXT_LOCALE")?.value;
  const browserLocale = detectLocale(headerStore.get("accept-language"));
  const locale = (cookieLocale && SUPPORTED_LOCALES.includes(cookieLocale))
    ? cookieLocale
    : browserLocale;

  return {
    locale,
    messages: (await import(`../../messages/${locale}.json`)).default,
  };
});
```

```typescript
"use client";

import { useEffect } from "react";
import { useAuthStore } from "@/lib/auth";

export function LocaleProvider({ children }: { children: React.ReactNode }) {
  const user = useAuthStore((s) => s.user);

  useEffect(() => {
    // When user logs in with a saved language preference, set cookie + reload if different
    if (user?.language) {
      const currentCookie = document.cookie
        .split("; ")
        .find((c) => c.startsWith("NEXT_LOCALE="))
        ?.split("=")[1];

      if (currentCookie !== user.language) {
        document.cookie = `NEXT_LOCALE=${user.language}; path=/; max-age=31536000; SameSite=Lax`;
        window.location.reload();
      }
    }
  }, [user?.language]);

  return <>{children}</>;
}
```

### Language selector component:

```typescript
"use client";

import { useTranslations } from "next-intl";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { useAuthStore } from "@/lib/auth";
import { apiClient } from "@/lib/api-client";
import { toast } from "sonner";

const LANGUAGES = [
  { code: "en", label: "English", flag: "🇬🇧" },
  { code: "pl", label: "Polski", flag: "🇵🇱" },
];

export function LanguageSelector() {
  const t = useTranslations("settings");
  const user = useAuthStore((s) => s.user);
  const currentLocale = useAuthStore((s) => s.locale);

  const handleChange = async (locale: string) => {
    try {
      if (user) {
        await apiClient("/v1/users/me", {
          method: "PATCH",
          body: JSON.stringify({ language: locale }),
        });
      }
      document.cookie = `NEXT_LOCALE=${locale}; path=/; max-age=31536000; SameSite=Lax`;
      window.location.reload();
    } catch {
      toast.error(t("languageSaveFailed"));
    }
  };

  return (
    <Select value={currentLocale} onValueChange={handleChange}>
      <SelectTrigger className="w-[140px]">
        <SelectValue />
      </SelectTrigger>
      <SelectContent>
        {LANGUAGES.map((lang) => (
          <SelectItem key={lang.code} value={lang.code}>
            {lang.flag} {lang.label}
          </SelectItem>
        ))}
      </SelectContent>
    </Select>
  );
}
```
